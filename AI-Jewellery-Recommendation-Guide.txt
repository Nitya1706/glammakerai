7.2 Create recommendation.py

Open:

backend/ai/recommendation.py

Add:

def recommend_jewellery(
    occasion,
    outfit,
    style
):

    occasion = occasion.lower()
    outfit = outfit.lower()
    style = style.lower()


    recommendations = []


    # Wedding
    if occasion == "wedding":

        if style == "traditional":

            recommendations = [
                "Kundan Necklace",
                "Jhumka Earrings",
                "Maang Tikka",
                "Bangles"
            ]

        elif style == "royal":

            recommendations = [
                "Polki Necklace",
                "Chandbali Earrings",
                "Maang Tikka",
                "Gold Bangles"
            ]

        else:

            recommendations = [
                "Gold Necklace",
                "Jhumka Earrings",
                "Bangles"
            ]


    # Mehndi
    elif occasion == "mehndi":

        recommendations = [
            "Floral Jewellery",
            "Jhumka Earrings",
            "Choker Necklace",
            "Colorful Bangles"
        ]


    # Haldi
    elif occasion == "haldi":

        recommendations = [
            "Floral Necklace",
            "Floral Earrings",
            "Minimal Bangles"
        ]


    # Sangeet
    elif occasion == "sangeet":

        recommendations = [
            "Statement Necklace",
            "Chandbali Earrings",
            "Designer Bracelet"
        ]


    # Reception
    elif occasion == "reception":

        recommendations = [
            "Diamond Necklace",
            "Drop Earrings",
            "Diamond Bracelet"
        ]


    # Party
    elif occasion == "party":

        recommendations = [
            "Statement Earrings",
            "Layered Necklace",
            "Fashion Bracelet"
        ]


    # Casual
    elif occasion == "casual":

        recommendations = [
            "Stud Earrings",
            "Minimal Necklace",
            "Simple Bracelet"
        ]


    # Default
    else:

        recommendations = [
            "Stud Earrings",
            "Simple Necklace",
            "Bracelet"
        ]


    # Outfit adjustment

    if "saree" in outfit:

        recommendations.append(
            "Traditional Saree Jewellery"
        )

    elif "lehenga" in outfit:

        recommendations.append(
            "Lehenga Choker Set"
        )

    elif "gown" in outfit:

        recommendations.append(
            "Elegant Pendant"
        )


    # Style adjustment

    if style == "minimal":

        recommendations = recommendations[:3]

    elif style == "modern":

        recommendations = [
            item for item in recommendations
            if "Floral" not in item
        ]


    # Remove duplicates

    recommendations = list(
        dict.fromkeys(recommendations)
    )


    return recommendations

7.4 Create AI API

Add this route to app.py:

@app.route(
    "/api/ai/recommend",
    methods=["POST"]
)
def ai_recommend():

    if "user_id" not in session:

        return jsonify({
            "success": False,
            "message": "Please login first"
        }), 401


    data = request.get_json()


    occasion = data.get(
        "occasion",
        ""
    )

    outfit = data.get(
        "outfit",
        ""
    )

    style = data.get(
        "style",
        ""
    )


    if not occasion:

        return jsonify({
            "success": False,
            "message":
                "Please select an occasion"
        }), 400


    recommendations = recommend_jewellery(
        occasion,
        outfit,
        style
    )


    recommendation_data = {

        "user_id":
            session["user_id"],

        "occasion":
            occasion,

        "outfit":
            outfit,

        "style":
            style,

        "recommendations":
            recommendations,

        "created_at":
            datetime.utcnow()
    }


    recommendation_collection.insert_one(
        recommendation_data
    )


    return jsonify({

        "success": True,

        "recommendations":
            recommendations
    })

7.6 Create JavaScript

Create:

frontend/js/ai-recommendation.js

Add:

const form =
    document.getElementById(
        "recommendationForm"
    );


const result =
    document.getElementById(
        "recommendationResult"
    );


form.addEventListener(
    "submit",
    async function(event) {

        event.preventDefault();


        const occasion =
            document.getElementById(
                "occasion"
            ).value;


        const outfit =
            document.getElementById(
                "outfit"
            ).value;


        const style =
            document.getElementById(
                "style"
            ).value;


        if (!occasion) {

            result.innerHTML =
                "<p>Please select an occasion.</p>";

            return;
        }


        result.innerHTML =
            "<p>AI is analyzing your style...</p>";


        try {

            const response =
                await fetch(
                    "/api/ai/recommend",
                    {

                        method: "POST",

                        headers: {
                            "Content-Type":
                                "application/json"
                        },

                        body: JSON.stringify({

                            occasion:
                                occasion,

                            outfit:
                                outfit,

                            style:
                                style
                        })
                    }
                );


            const data =
                await response.json();


            if (!data.success) {

                result.innerHTML =
                    `<p>${data.message}</p>`;

                return;
            }


            let html =
                "<h3>✨ AI Recommendations</h3>";


            html += "<ul>";


            data.recommendations.forEach(
                function(item) {

                    html +=
                        `<li>${item}</li>`;
                }
            );


            html += "</ul>";


            result.innerHTML = html;


        } catch (error) {

            console.error(error);

            result.innerHTML =
                "<p>AI recommendation failed.</p>";
        }

    }
);
