8.3 Add image analysis code

Put this inside outfit_analysis.py:

import cv2


def analyze_outfit(image_path):

    image = cv2.imread(image_path)

    if image is None:
        return {
            "success": False,
            "message": "Unable to read image"
        }


    # Resize image
    image = cv2.resize(
        image,
        (300, 300)
    )


    # Calculate average BGR color
    average_color = image.mean(
        axis=(0, 1)
    )


    blue = average_color[0]
    green = average_color[1]
    red = average_color[2]


    # Detect dominant color
    if red > 180 and green > 180 and blue > 180:

        color = "White"

    elif red > green * 1.3 and red > blue * 1.3:

        color = "Red"

    elif green > red * 1.2 and green > blue * 1.2:

        color = "Green"

    elif blue > red * 1.2 and blue > green * 1.2:

        color = "Blue"

    elif red > 120 and green > 80 and blue < 100:

        color = "Yellow"

    elif red > 100 and blue > 80:

        color = "Pink/Purple"

    else:

        color = "Mixed"


    # Simple brightness calculation
    gray = cv2.cvtColor(
        image,
        cv2.COLOR_BGR2GRAY
    )

    brightness = gray.mean()


    if brightness > 180:

        style = "Light / Elegant"

    elif brightness < 80:

        style = "Dark / Bold"

    else:

        style = "Balanced"


    return {

        "success": True,

        "color": color,

        "style": style,

        "brightness": round(
            float(brightness),
            2
        ),

        "message":
            "Outfit analyzed successfully"
    }
What this does

The program reads the uploaded image and calculates:

 ↓
Brightness

Style

8.5 Create AI analysis API

Add this route:

@app.route(
    "/api/ai/analyze-outfit",
    methods=["POST"]
)
def analyze_outfit_api():

    if "user_id" not in session:

        return jsonify({
            "success": False,
            "message": "Please login first"
        }), 401


    outfit = request.files.get("outfit")


    if not outfit:

        return jsonify({
            "success": False,
            "message":
                "Please upload an outfit image"
        }), 400


    if not allowed_file(outfit.filename):

        return jsonify({
            "success": False,
            "message":
                "Invalid image format"
        }), 400


    filename = (
        str(uuid.uuid4())
        + "_"
        + secure_filename(
            outfit.filename
        )
    )


    filepath = os.path.join(
        OUTFIT_FOLDER,
        filename
    )


    outfit.save(filepath)


    analysis = analyze_outfit(
        filepath
    )


    if not analysis["success"]:

        return jsonify(analysis), 400


    analysis_data = {

        "user_id":
            session["user_id"],

        "image":
            filename,

        "color":
            analysis["color"],

        "style":
            analysis["style"],

        "brightness":
            analysis["brightness"],

        "created_at":
            datetime.utcnow()
    }


    recommendation_collection.insert_one(
        analysis_data
    )


    return jsonify({

        "success": True,

        "analysis": analysis
    })

8.7 Create JavaScript

Create:

frontend/js/outfit-analysis.js

Add:

const form =
    document.getElementById(
        "analysisForm"
    );


const imageInput =
    document.getElementById(
        "outfitImage"
    );


const preview =
    document.getElementById(
        "preview"
    );


const result =
    document.getElementById(
        "analysisResult"
    );


// Image preview

imageInput.addEventListener(
    "change",
    function() {

        const file =
            imageInput.files[0];

        if (!file) {
            return;
        }


        const imageURL =
            URL.createObjectURL(file);


        preview.innerHTML = `

            <br>

            <img
                src="${imageURL}"
                width="300"
                alt="Outfit Preview"
            >

        `;
    }
);


// Submit

form.addEventListener(
    "submit",
    async function(event) {

        event.preventDefault();


        const file =
            imageInput.files[0];


        if (!file) {

            result.innerHTML =
                "<p>Please select an image.</p>";

            return;
        }


        const formData =
            new FormData();


        formData.append(
            "outfit",
            file
        );


        result.innerHTML =
            "<p>🤖 AI is analyzing your outfit...</p>";


        try {

            const response =
                await fetch(
                    "/api/ai/analyze-outfit",
                    {
                        method: "POST",
                        body: formData
                    }
                );


            const data =
                await response.json();


            if (!data.success) {

                result.innerHTML =
                    `<p>${data.message}</p>`;

                return;
            }


            const analysis =
                data.analysis;


            result.innerHTML = `

                <h2>
                    ✨ AI Analysis Result
                </h2>

                <p>
                    <strong>Dominant Color:</strong>
                    ${analysis.color}
                </p>

                <p>
                    <strong>Style:</strong>
                    ${analysis.style}
                </p>

                <p>
                    <strong>Brightness:</strong>
                    ${analysis.brightness}
                </p>

                <p>
                    ${analysis.message}
                </p>

            `;


        } catch (error) {

            console.error(error);

            result.innerHTML =
                "<p>AI analysis failed.</p>";
        }

    }
);
