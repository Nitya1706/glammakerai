9.2 Add occasion prediction

Put this code in style_matcher.py:

def predict_occasion(outfit_color, outfit_style):

    color = outfit_color.lower()
    style = outfit_style.lower()

    # Dark and bold outfits
    if "dark" in style:

        if color in ["red", "purple", "blue"]:
            return "Reception"

        return "Party"

    # Light outfits
    if "light" in style:

        if color in ["yellow", "pink", "white"]:
            return "Haldi"

        return "Casual"

    # Balanced outfits
    if "balanced" in style:

        if color in ["red", "maroon", "gold"]:
            return "Wedding"

        if color in ["green", "pink"]:
            return "Mehndi"

        return "Sangeet"

    return "Casual"

This takes the information from Step 8 and predicts a likely occasion.

9.3 Add jewellery matching

Below the previous function, add:

def match_jewellery(
    occasion,
    outfit_color,
    outfit_style
):

    occasion = occasion.lower()
    color = outfit_color.lower()

    if occasion == "wedding":

        jewellery = [
            "Kundan Necklace",
            "Jhumka Earrings",
            "Maang Tikka",
            "Gold Bangles"
        ]

    elif occasion == "mehndi":

        jewellery = [
            "Floral Jewellery",
            "Colorful Bangles",
            "Jhumka Earrings"
        ]

    elif occasion == "haldi":

        jewellery = [
            "Floral Necklace",
            "Floral Earrings",
            "Minimal Bangles"
        ]

    elif occasion == "sangeet":

        jewellery = [
            "Statement Necklace",
            "Chandbali Earrings",
            "Designer Bracelet"
        ]

    elif occasion == "reception":

        jewellery = [
            "Diamond Necklace",
            "Drop Earrings",
            "Diamond Bracelet"
        ]

    elif occasion == "party":

        jewellery = [
            "Statement Earrings",
            "Layered Necklace",
            "Fashion Bracelet"
        ]

    else:

        jewellery = [
            "Stud Earrings",
            "Minimal Necklace",
            "Simple Bracelet"
        ]


    # Color matching

    if color == "red":

        jewellery.append(
            "Gold Jewellery"
        )

    elif color == "blue":

        jewellery.append(
            "Silver Jewellery"
        )

    elif color == "green":

        jewellery.append(
            "Gold and Emerald Jewellery"
        )

    elif color == "white":

        jewellery.append(
            "Diamond Jewellery"
        )

    elif color == "yellow":

        jewellery.append(
            "Gold Jewellery"
        )


    return list(
        dict.fromkeys(jewellery)
    )
9.4 Create the complete AI styling function

At the bottom of style_matcher.py add:

def generate_style_match(
    outfit_color,
    outfit_style
):

    occasion = predict_occasion(
        outfit_color,
        outfit_style
    )

    jewellery = match_jewellery(
        occasion,
        outfit_color,
        outfit_style
    )

    return {
        "occasion": occasion,
        "jewellery": jewellery
    }

Add this route:

@app.route(
    "/api/ai/style-match",
    methods=["POST"]
)
def ai_style_match():

    if "user_id" not in session:

        return jsonify({
            "success": False,
            "message": "Please login first"
        }), 401


    outfit = request.files.get("outfit")


    if not outfit:

        return jsonify({
            "success": False,
            "message":
                "Please upload an outfit image"
        }), 400


    if not allowed_file(outfit.filename):

        return jsonify({
            "success": False,
            "message":
                "Invalid image format"
        }), 400


    filename = (
        str(uuid.uuid4())
        + "_"
        + secure_filename(
            outfit.filename
        )
    )


    filepath = os.path.join(
        OUTFIT_FOLDER,
        filename
    )


    outfit.save(filepath)


    # Step 8 AI analysis
    analysis = analyze_outfit(
        filepath
    )


    if not analysis["success"]:

        return jsonify(analysis), 400


    # Step 9 AI style matching
    style_match = generate_style_match(

        analysis["color"],

        analysis["style"]
    )


    # Save complete result
    style_data = {

        "user_id":
            session["user_id"],

        "image":
            filename,

        "color":
            analysis["color"],

        "style":
            analysis["style"],

        "brightness":
            analysis["brightness"],

        "occasion":
            style_match["occasion"],

        "jewellery":
            style_match["jewellery"],

        "created_at":
            datetime.utcnow()
    }


    recommendation_collection.insert_one(
        style_data
    )


    return jsonify({

        "success": True,

        "analysis": {

            "color":
                analysis["color"],

            "style":
                analysis["style"],

            "brightness":
                analysis["brightness"]
        },

        "occasion":
            style_match["occasion"],

        "jewellery":
            style_match["jewellery"]
    })

const form =
    document.getElementById(
        "stylistForm"
    );


const imageInput =
    document.getElementById(
        "stylistImage"
    );


const preview =
    document.getElementById(
        "stylistPreview"
    );


const result =
    document.getElementById(
        "stylistResult"
    );


// Preview image

imageInput.addEventListener(
    "change",
    function() {

        const file =
            imageInput.files[0];

        if (!file) {
            return;
        }

        const imageURL =
            URL.createObjectURL(file);

        preview.innerHTML = `

            <br>

            <img
                src="${imageURL}"
                width="300"
                alt="Outfit Preview"
            >

        `;
    }
);


// AI analysis

form.addEventListener(
    "submit",
    async function(event) {

        event.preventDefault();


        const file =
            imageInput.files[0];


        if (!file) {

            result.innerHTML =
                "<p>Please select an image.</p>";

            return;
        }


        const formData =
            new FormData();


        formData.append(
            "outfit",
            file
        );


        result.innerHTML = `
            <p>
                🤖 AI is analyzing your outfit...
            </p>
        `;


        try {

            const response =
                await fetch(
                    "/api/ai/style-match",
                    {
                        method: "POST",
                        body: formData
                    }
                );


            const data =
                await response.json();


            if (!data.success) {

                result.innerHTML =
                    `<p>${data.message}</p>`;

                return;
            }


            let jewelleryHTML = "";


            data.jewellery.forEach(
                function(item) {

                    jewelleryHTML += `
                        <li>${item}</li>
                    `;

                }
            );


            result.innerHTML = `

                <div class="ai-result">

                    <h2>
                        ✨ AI Style Result
                    </h2>

                    <p>
                        <strong>
                            Outfit Color:
                        </strong>

                        ${data.analysis.color}
                    </p>


                    <p>
                        <strong>
                            Outfit Style:
                        </strong>

                        ${data.analysis.style}
                    </p>


                    <p>
                        <strong>
                            Suggested Occasion:
                        </strong>

                        ${data.occasion}
                    </p>


                    <h3>
                        💎 Recommended Jewellery
                    </h3>


                    <ul>
                        ${jewelleryHTML}
                    </ul>

                </div>

            `;


        } catch (error) {

            console.error(error);

            result.innerHTML =
                "<p>AI analysis failed.</p>";
        }

    }
);
