AI Stylist Chatbot

In Step 9, GlamMaker can analyze an outfit and recommend jewellery. Now we will add a chatbot where the user can ask questions like:

"What jewellery should I wear with a red saree?"
"Suggest jewellery for a wedding."
"What should I wear for Haldi?"
"Which jewellery is best with blue dress?"
"Give me a simple look for reception."

For now, we will make it local and beginner-friendly, without requiring an external paid AI API.

10.1 Create AI chatbot file

Create:

backend/
└── ai/
    └── chatbot.py

Add:

def get_stylist_response(message):
    message = message.lower().strip()

    # Wedding
    if "wedding" in message:
        return {
            "reply": "For a wedding, try Kundan or Polki jewellery with Jhumka earrings, Maang Tikka and Gold Bangles."
        }

    # Mehndi
    if "mehndi" in message:
        return {
            "reply": "For Mehndi, floral jewellery, colorful bangles and Jhumka earrings are a great choice."
        }

    # Haldi
    if "haldi" in message:
        return {
            "reply": "For Haldi, choose light and colorful floral jewellery with a simple necklace and earrings."
        }

    # Sangeet
    if "sangeet" in message:
        return {
            "reply": "For Sangeet, try statement jewellery, Chandbali earrings and a designer bracelet."
        }

    # Reception
    if "reception" in message:
        return {
            "reply": "For a reception, Diamond jewellery, Drop earrings and a Diamond bracelet will give an elegant look."
        }

    # Red outfit
    if "red" in message:
        return {
            "reply": "For a red outfit, Gold jewellery is a beautiful match. Try a Gold necklace and Jhumka earrings."
        }

    # Blue outfit
    if "blue" in message:
        return {
            "reply": "For a blue outfit, Silver or Diamond jewellery works very well. Try Drop earrings with a simple necklace."
        }

    # Green outfit
    if "green" in message:
        return {
            "reply": "For a green outfit, Gold and Emerald jewellery can create a beautiful traditional look."
        }

    # Saree
    if "saree" in message:
        return {
            "reply": "With a saree, try a traditional necklace, Jhumka earrings, bangles and a Maang Tikka."
        }

    # Lehenga
    if "lehenga" in message:
        return {
            "reply": "With a lehenga, a Choker necklace, Chandbali earrings and Maang Tikka are excellent choices."
        }

    # Casual
    if "casual" in message:
        return {
            "reply": "For a casual look, keep it simple with Stud earrings, a Minimal necklace and a small bracelet."
        }

    # Greeting
    if "hello" in message or "hi" in message or "hey" in message:
        return {
            "reply": "Hello! 👋 I am your GlamMaker AI Stylist. Tell me about your outfit or occasion."
        }

    # Help
    if "help" in message:
        return {
            "reply": "You can ask me about jewellery for Wedding, Mehndi, Haldi, Sangeet, Reception, Saree, Lehenga or different outfit colors."
        }

    return {
        "reply": "I can help you choose jewellery. Tell me your outfit color, outfit type or occasion."
    }
10.2 Add chatbot API to app.py

At the top of app.py, add:

from ai.chatbot import get_stylist_response

Then add:

@app.route("/api/ai/chat", methods=["POST"])
def ai_chat():

    if "user_id" not in session:
        return jsonify({
            "success": False,
            "message": "Please login first"
        }), 401

    data = request.get_json()

    message = data.get("message", "").strip()

    if not message:
        return jsonify({
            "success": False,
            "message": "Please enter a message"
        }), 400

    result = get_stylist_response(message)

    return jsonify({
        "success": True,
        "reply": result["reply"]
    })

10.4 Create JavaScript

Create:

frontend/js/ai-stylist-chat.js

Add:

const chatForm = document.getElementById("chatForm");
const messageInput = document.getElementById("message");
const chatBox = document.getElementById("chatBox");


chatForm.addEventListener("submit", async function(event) {

    event.preventDefault();

    const message = messageInput.value.trim();

    if (!message) {
        return;
    }


    // Show user message

    const userMessage = document.createElement("div");

    userMessage.className = "user-message";

    userMessage.textContent = message;

    chatBox.appendChild(userMessage);


    messageInput.value = "";


    try {

        const response = await fetch("/api/ai/chat", {

            method: "POST",

            headers: {
                "Content-Type": "application/json"
            },

            body: JSON.stringify({
                message: message
            })

        });


        const data = await response.json();


        const botMessage = document.createElement("div");

        botMessage.className = "bot-message";


        if (data.success) {

            botMessage.textContent = data.reply;

        } else {

            botMessage.textContent =
                data.message || "Something went wrong.";

        }


        chatBox.appendChild(botMessage);


        chatBox.scrollTop = chatBox.scrollHeight;


    } catch (error) {

        console.error(error);

        const errorMessage =
            document.createElement("div");

        errorMessage.className =
            "bot-message";

        errorMessage.textContent =
            "Unable to connect to AI Stylist.";

        chatBox.appendChild(errorMessage);

    }

});
